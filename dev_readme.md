-- =====================================================
-- 📦 TABLE: songs (USER UPLOADED SONGS)
-- =====================================================
CREATE TABLE public.songs (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT now(),
    title TEXT NULL,
    song_path TEXT NULL,
    author TEXT NULL,
    user_id UUID NOT NULL,  -- 👈 ENFORCES OWNERSHIP
    image_path TEXT NULL,
    CONSTRAINT songs_pkey PRIMARY KEY (id),
    CONSTRAINT songs_user_id_fkey FOREIGN KEY (user_id) REFERENCES auth.users (id) ON UPDATE CASCADE ON DELETE CASCADE
) TABLESPACE pg_default;

-- 🔐 RLS POLICIES FOR songs
ALTER TABLE public.songs ENABLE ROW LEVEL SECURITY;

DO $$
BEGIN
    -- PUBLIC READ ACCESS (EVERYONE CAN SEE SONGS)
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename = 'songs' AND policyname = 'Allow public read access') THEN
        CREATE POLICY "Allow public read access" 
        ON public.songs FOR SELECT USING (true);
    END IF;

    -- AUTHENTICATED USERS CAN INSERT THEIR OWN SONGS
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename = 'songs' AND policyname = 'Allow users to insert their own songs') THEN
        CREATE POLICY "Allow users to insert their own songs" 
        ON public.songs FOR INSERT WITH CHECK (auth.uid() = user_id);
    END IF;

    -- OPTIONAL: ALLOW USERS TO UPDATE THEIR OWN SONGS
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename = 'songs' AND policyname = 'Allow users to update their own songs') THEN
        CREATE POLICY "Allow users to update their own songs" 
        ON public.songs FOR UPDATE USING (auth.uid() = user_id);
    END IF;
END $$;

# SQL for buckets

-- =====================================================
-- 🪣 CREATE BUCKETS IF NOT EXISTS
-- =====================================================

-- 📦 Bucket: images (public)
INSERT INTO storage.buckets (id, name, public)
SELECT 'images', 'images', true
WHERE NOT EXISTS (
  SELECT 1 FROM storage.buckets WHERE name = 'images'
);

-- 📦 Bucket: songs (public) with allowed MIME type
INSERT INTO storage.buckets (id, name, public)
SELECT 'songs', 'songs', true
WHERE NOT EXISTS (
  SELECT 1 FROM storage.buckets WHERE name = 'songs'
);

-- 🎵 Set allowed MIME types for songs bucket (only audio/mpeg)
UPDATE storage.buckets
SET allowed_mime_types = ARRAY['audio/mpeg']
WHERE name = 'songs' AND (allowed_mime_types IS NULL OR allowed_mime_types != ARRAY['audio/mpeg']);



-- =====================================================
-- 🔓 BUCKET-LEVEL CRUD POLICIES (GLOBAL)
-- =====================================================
-- These apply to the storage.buckets table itself (managing buckets)

DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Allow bucket creation') THEN
    CREATE POLICY "Allow bucket creation" ON storage.buckets
      FOR INSERT WITH CHECK (true);
  END IF;
END $$;

DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Allow bucket read') THEN
    CREATE POLICY "Allow bucket read" ON storage.buckets
      FOR SELECT USING (true);
  END IF;
END $$;

DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Allow bucket update') THEN
    CREATE POLICY "Allow bucket update" ON storage.buckets
      FOR UPDATE USING (true);
  END IF;
END $$;

DO $$
BEGIN
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Allow bucket deletion') THEN
    CREATE POLICY "Allow bucket deletion" ON storage.buckets
      FOR DELETE USING (true);
  END IF;
END $$;



-- =====================================================
-- 🖼️ OBJECT-LEVEL POLICIES FOR BUCKET: images
-- =====================================================
-- SELECT: everyone
-- INSERT: authenticated users only
-- UPDATE: only owner (auth.uid() = owner)
-- DELETE: only owner

DO $$
BEGIN
  -- SELECT
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename = 'objects' AND policyname = 'Allow public read on images') THEN
    CREATE POLICY "Allow public read on images" ON storage.objects
      FOR SELECT USING (bucket_id = 'images');
  END IF;

  -- INSERT
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename = 'objects' AND policyname = 'Allow authenticated insert on images') THEN
    CREATE POLICY "Allow authenticated insert on images" ON storage.objects
      FOR INSERT WITH CHECK (bucket_id = 'images' AND auth.role() = 'authenticated');
  END IF;

  -- UPDATE
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename = 'objects' AND policyname = 'Allow owner update on images') THEN
    CREATE POLICY "Allow owner update on images" ON storage.objects
      FOR UPDATE USING (bucket_id = 'images' AND auth.uid() = owner);
  END IF;

  -- DELETE
  IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE tablename = 'objects' AND policyname = 'Allow owner delete on images') THEN
    CREATE POLICY "Allow owner delete on images" ON storage.objects
      FOR DELETE USING (bucket_id = 'images' AND auth.uid() = owner);
  END IF;
END $$;



-- =====================================================
-- 🎵 OBJECT-LEVEL POLICIES FOR BUCKET: songs
